16"""Define the MySQL Backups plugin's self-contained catalog model and API response. 

17 

18This module is intentionally **self-contained** — it imports only from 

19``app.core``, ``pydantic``, ``sqlalchemy``, ``sqlmodel``, and ``yaml`` (for the 

20shared config parser), never from ``app.inventory`` / ``app.tasks`` / the app 

21framework's form DSL. The sep Alembic discovery loads it at migration time to 

22register the table in the migration metadata, and pulling in those heavier 

23modules would bleed their tables into the sep autogenerate comparison and break 

24``make checkmigrations``. 

25 

26The split mirrors ``app.sep.apps.atw``, in the direction that matters: there, 

27``atw.models`` is the self-contained module and the one inventory-dependent 

28piece (the category taxonomy) lives apart, in ``atw.categories``. Here, this 

29module plays the role of ``atw.models``, and the plugin's heavy form/DSL 

30surface (:class:`~app.sep.apps.mysql_backups.forms.BackupCreate` and friends) 

31plays the role of ``atw.categories`` — the piece split *out* because it, not 

32the table, needed the heavier imports. 

33"""

57class MysqlBackupRun(BaseSQLModel, table=True): 

58 """Persist one completed MySQL backup run's produced output. 

59 

60 Written by the run-result recorder on a successful mydumper or xtrabackup 

61 run — one row per run, keyed by :attr:`task_history_id`. The record is 

62 tool-agnostic: the xtrabackup incremental layout is captured as a different 

63 :attr:`location` string, not a different shape. Binlog runs and non-success 

64 terminals leave no row. 

65 

66 :param task_history_id: The id of the ``TaskHistory`` this run belongs to; 

67 unique, so one run maps to exactly one record. 

68 :param service_name: The inventory service name the backup was taken from 

69 (the task's ``_service_name`` meta), kept as the fallback query key for 

70 rows carrying no :attr:`service_id` and as the only usable key for a 

71 free-typed restore destination that has no inventory row at all. A row 

72 reached this way can still be confused with another service's: two MySQL 

73 services may share a name, since ``Service.name`` carries no uniqueness 

74 constraint. 

75 :param service_id: The inventory id of the service the backup was taken from 

76 (the task's ``_service_id`` meta), preferred as the per-service query key 

77 because it survives a rename, which the name does not. Not a foreign key 

78 — the catalog is sep-owned and inventory lives in a separate database, so 

79 nothing enforces that the id still resolves. Empty on rows recorded 

80 before the id was stamped, which is what keeps the name fallback needed. 

81 :param hostname: The backup target host (the task's ``target`` meta). 

82 :param backup_type: The backup tool the run used, mydumper or xtrabackup. 

83 :param location: The resolved on-disk directory the run produced, stored 

84 exactly as the payload reported it. 

85 :param upload_destination: The upload destination when one was configured, 

86 else ``None``. 

87 :param size_bytes: The backup size in bytes, when the run reported it. 

88 :param started_at: When the run started. 

89 :param finished_at: When the run finished. 

90 """ 

91 

92 __tablename__ = "mysql_backup_run"

180def extract_backup_type_marker(task_data: dict[str, Any] | None) -> str | None: 

181 """Return the ``BACKUP_TYPE`` marker from a task's YAML config, or ``None``. 

182 

183 Reads the value defensively: a missing ``meta``, unparseable YAML, or an 

184 absent key all resolve to ``None`` rather than raising. Returns the raw 

185 single-letter marker (``"M"``/``"X"``/``"B"``); callers that need the typed 

186 :class:`BackupType` coerce it themselves (see ``deps.py``'s 

187 ``_extract_backup_type_from_task``). 

188 

189 :param task_data: The task's ``data`` dict (``meta`` carries the YAML config). 

190 :return: The raw backup-type marker, or ``None``. 

191 """ 

192 match task_data: 

193 case {"meta": {"config": str(raw_config)}} if raw_config: 

194 pass 

195 case _: 

196 return None 

197 

198 try: 

199 config = yaml.safe_load(raw_config) 

200 except yaml.YAMLError: 

201 return None 

202 

203 match config: 

204 case {"SERVER_LIST": [{"BACKUP_TYPE": str(backup_type)}, *_]}: 

205 return backup_type 

206 case _: 

207 return None
