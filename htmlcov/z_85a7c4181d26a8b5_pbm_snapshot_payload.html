16def _backup_user_home() -> Path | None:

18 

19 Resolve the user the MySQL/PostgreSQL payloads resolve -- 

20 ``PERCONA_BACKUP_USER``, then ``SUDO_USER``, then the login name -- and read 

21 the home from the passwd database rather than ``$HOME``. Under ``sudo`` the 

22 two disagree, and honouring ``$HOME`` would drop this payload's ``.prom`` 

23 files somewhere PMM never scrapes while the MySQL/PG ones land correctly. 

24 

25 Unlike those payloads the lookup is lazy and guarded: a host with no passwd 

26 entry for the resolved name must cost the metric, not abort the backup. 

27 

28 :return: The backup user's home directory, or ``None`` when it cannot be 

29 resolved to an absolute path. 

30 """

32 user = ( 

33 os.environ.get("PERCONA_BACKUP_USER") 

34 or os.environ.get("SUDO_USER") 

35 or getpass.getuser() 

36 ) 

37 home = Path(pwd.getpwnam(user).pw_dir)

75def _escape_label_value(value: str) -> str: 

76 r"""Return ``value`` escaped for use as a Prometheus label value. 

77 

78 The text-exposition format defines exactly three escapes inside a label value 

79 -- backslash, double-quote and newline -- and its parser rejects any other 

80 escape sequence. An un-escaped alias carrying one of those three would produce 

81 a ``.prom`` file the collector rejects wholesale, so escape them and leave 

82 every other character raw: a carriage return is copied through by the parser, 

83 while ``\r`` would be the invalid escape that breaks the file. 

84 

85 :param value: The raw label value (the host alias). 

86 :return: The escaped label value. 

87 """ 

88 return value.replace("\\", "\\\\").replace('"', '\\"').replace("\n", "\\n")
