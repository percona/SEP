117async def fire_change_callbacks( 

118 callbacks: CallbackRegistry, 

119 setting_class: SettingClassEnum, 

120 previous: Mapping[str, object], 

121 current: Mapping[str, object], 

122) -> None: 

123 """Fire the registered callback for every key whose snapshot value changed. 

124 

125 Compares ``previous`` against ``current`` and, for each ``(setting_class, 

126 key)`` whose value differs and has a registered callback, awaits the 

127 callback. Each callback runs inside its own ``try/except`` so one failure 

128 neither aborts the cycle nor blocks the remaining callbacks. 

129 

130 Shared by the background refresher (:func:`refresh_all`, diffing the snapshot 

131 it just rebuilt) and the settings-API PATCH/DELETE handlers (diffing the 

132 snapshot they publish inline), so both publish paths deliver the same rebind 

133 notifications. 

134 

135 :param callbacks: The registered rebind callbacks keyed by 

136 ``(setting_class, key)``. 

137 :type callbacks: CallbackRegistry 

138 :param setting_class: The class whose snapshot was just republished. 

139 :type setting_class: SettingClassEnum 

140 :param previous: The snapshot in effect before the republish. 

141 :type previous: Mapping[str, object] 

142 :param current: The snapshot now in effect. 

143 :type current: Mapping[str, object] 

144 """ 

145 for key in previous.keys() | current.keys(): 

146 if previous.get(key) == current.get(key): 

147 continue 

148 callback = callbacks.get((setting_class, key)) 

149 if callback is None: 

150 continue 

151 try: 

152 await callback(current) 

153 except Exception: 

154 logger.exception( 

155 "Rebind callback for %s.%s failed; keeping previous binding", 

156 setting_class.name, 

157 key, 

158 )

161async def refresh_all( 

162 session_maker_factory: SessionMakerFactory, 

163 proxies: ProxyRegistry, 

164 callbacks: CallbackRegistry | None = None, 

165) -> None: 

166 """Refresh override snapshots for all wired proxies in a single session. 

167 

168 Per-proxy failures (raised from :func:`build_snapshot` once a session is 

169 open) are caught and logged: the previous snapshot is retained for that 

170 proxy and the rest of ``proxies`` continue to refresh. The session is 

171 rolled back after each failure so that on Postgres -- where a failed 

172 query leaves the transaction in ``InFailedSqlTransaction`` state until 

173 explicit rollback -- the next proxy's ``manager.list(...)`` does not 

174 inherit the aborted transaction. Failures that occur *before* per-proxy 

175 iteration begins -- specifically from ``session_maker_factory()`` or 

176 ``async_session_maker()`` -- are NOT handled here; they propagate to the 

177 caller. The background refresher in :func:`start_refresh_task` wraps its 

178 periodic invocation in a broad ``except`` so transient engine/pool 

179 errors do not kill the task. 

180 

181 When ``callbacks`` is supplied, the snapshot in effect before each proxy's 

182 republish is diffed against the new one and the registered callback for any 

183 changed ``(setting_class, key)`` is fired (see :func:`fire_change_callbacks`). 

184 A proxy whose republish failed is skipped without firing callbacks. The 

185 initial inline refresh in :func:`start_refresh_task` passes no callbacks, so 

186 startup seeding never triggers a rebind. 

187 

188 :param session_maker_factory: A zero-argument callable returning a 

189 service-scoped ``async_sessionmaker``. Invoked exactly once per call 

190 to avoid recreating session makers on each iteration. 

191 :type session_maker_factory: SessionMakerFactory 

192 :param proxies: The wired proxy registry keyed by class identifier. 

193 :type proxies: ProxyRegistry 

194 :param callbacks: Optional rebind callbacks fired for changed keys. When 

195 ``None``, snapshots are republished without any change detection. 

196 :type callbacks: CallbackRegistry | None 

197 :raises Exception: Re-raises any failure from 

198 ``session_maker_factory()`` itself (e.g. the factory is misconfigured 

199 or its engine cannot be constructed). Connection-time failures 

200 against the DB -- engine unreachable, pool exhausted, auth -- do not 

201 surface here because ``async_session_maker()`` does not check out a 

202 connection; those errors fire later inside ``build_snapshot()`` and 

203 are caught by the per-proxy handler. 

204 """ 

205 async_session_maker = session_maker_factory() 

206 async with async_session_maker() as session: 

207 for setting_class, entry in proxies.items(): 

208 previous = entry.proxy.get_snapshot() if callbacks else None 

209 try: 

210 await publish_snapshot(entry.proxy, session, entry.settings_cls) 

211 except Exception: 

212 logger.exception( 

213 "Failed to refresh overrides for %s; keeping previous snapshot", 

214 setting_class.name, 

215 ) 

216 # Roll back so a Postgres ``InFailedSqlTransaction`` from this 

217 # proxy does not cascade into every subsequent proxy's 

218 # ``manager.list(...)`` on the shared session. 

219 await session.rollback() 

220 continue 

221 if callbacks is not None and previous is not None: 

222 await fire_change_callbacks( 

223 callbacks, setting_class, previous, entry.proxy.get_snapshot() 

224 )

227async def start_refresh_task( 

228 session_maker_factory: SessionMakerFactory, 

229 proxies: ProxyRegistry, 

230 interval: timedelta, 

231 callbacks: CallbackRegistry | None = None, 

232 *, 

233 seed_timeout: float | None = None, 

234) -> asyncio.Task: 

235 """Perform an initial refresh and start a background refresh loop. 

236 

237 The initial refresh awaits inline so the lifespan does not yield until 

238 the first snapshot has been observed. Subsequent refreshes run inside an 

239 :func:`asyncio.create_task` that sleeps for ``interval`` between cycles. 

240 

241 ``callbacks`` are passed only to the periodic loop's :func:`refresh_all`, 

242 never to the inline initial refresh -- the startup snapshot seeds the 

243 proxies without firing rebind callbacks (long-lived objects are constructed 

244 against the effective snapshot directly during lifespan startup). 

245 

246 When ``seed_timeout`` is set, the inline seed is bounded by 

247 :func:`asyncio.wait`. On expiry the seed task is cancelled without 

248 awaiting its unwind -- so a hung ``AsyncSession.__aexit__`` cannot push 

249 the child past the budget -- the timeout is logged at ERROR, and the 

250 periodic refresher is still created, so the child starts with unseeded 

251 (env-only) overrides rather than without a refresher. When 

252 ``seed_timeout`` is ``None`` the seed awaits unbounded, matching the 

253 historical behaviour used by the web lifespans. 

254 

255 :param session_maker_factory: A zero-argument callable returning a 

256 service-scoped ``async_sessionmaker``. 

257 :type session_maker_factory: SessionMakerFactory 

258 :param proxies: The wired proxy registry keyed by class identifier. 

259 :type proxies: ProxyRegistry 

260 :param interval: The wall-clock delay between refresh cycles. Must be a 

261 positive duration; the :class:`Settings` field validator enforces 

262 this at construction time. 

263 :type interval: timedelta 

264 :param callbacks: Optional rebind callbacks fired by the periodic loop when 

265 a watched override changes. Not applied to the initial refresh. 

266 :type callbacks: CallbackRegistry | None 

267 :param seed_timeout: Optional wall-clock budget in seconds for the inline 

268 seed. ``None`` (the default) leaves the seed unbounded. 

269 :return: The background refresh task. Callers must cancel and await this 

270 task during shutdown to drain pending iterations cleanly. 

271 :rtype: asyncio.Task 

272 :raises Exception: Re-raises any failure from the inline initial 

273 :func:`refresh_all` call -- in practice limited to 

274 ``session_maker_factory()`` failures (see :func:`refresh_all` for the 

275 narrowed contract). Connection-time DB failures from the initial 

276 snapshot build are caught per-proxy and do NOT propagate. A 

277 ``seed_timeout`` expiry is caught here and does NOT propagate; the 

278 periodic task is still created. Subsequent iterations inside the 

279 background task are also wrapped in ``except`` and do NOT propagate; 

280 only ``session_maker_factory()`` failures at startup can break the 

281 lifespan. 

282 """ 

283 if seed_timeout is None: 

284 await refresh_all(session_maker_factory, proxies) 

285 else: 

286 # Prefer ``asyncio.wait`` over ``wait_for``: ``wait_for`` awaits the 

287 # cancelled coroutine's unwind, so a hung ``AsyncSession.__aexit__`` 

288 # (returning a stuck connection to the pool) would still blow the 

289 # budget. Cancel without awaiting so the wall-clock bound holds. 

290 seed_task = asyncio.create_task(refresh_all(session_maker_factory, proxies)) 

291 done, _ = await asyncio.wait({seed_task}, timeout=seed_timeout) 

292 if done: 

293 await seed_task 

294 else: 

295 seed_task.cancel() 

296 seed_task.add_done_callback(_drain_cancelled_seed_task) 

297 logger.error( 

298 "Initial settings-override refresh exceeded its %.2fs seed " 

299 "budget; starting the periodic refresher with unseeded " 

300 "overrides", 

301 seed_timeout, 

302 ) 

303 interval_seconds = interval.total_seconds() 

304 

305 async def _loop() -> None: 

306 while True: 

307 await asyncio.sleep(interval_seconds) 

308 try: 

309 await refresh_all(session_maker_factory, proxies, callbacks) 

310 except Exception: 

311 logger.exception( 

312 "Settings override refresher iteration failed; will retry next cycle" 

313 ) 

314 

315 return asyncio.create_task(_loop(), name="settings-override-refresher")

318def resolve_refresher_options( 

319 interval: timedelta | None, *, enabled: bool | None 

320) -> tuple[timedelta, bool]: 

321 """Return the refresher options, reading settings for whichever is unset. 

322 

323 ``Settings.SETTINGS_OVERRIDE`` configures the refresher for all three 

324 services, so every start-up site would otherwise repeat the same two reads 

325 and each is a place the next rename can miss one. Reading here also reads 

326 late enough to see a value the proxy snapshot has since refreshed. 

327 

328 :param interval: The caller's explicit refresh interval, or ``None``. 

329 :param enabled: The caller's explicit enabled flag, or ``None``. 

330 :return: ``(interval, enabled)`` with any ``None`` replaced by the 

331 configured value. 

332 """ 

333 if interval is not None and enabled is not None: 

334 return interval, enabled 

335 # circular import: config imports registry imports policy imports this package 

336 from app.core.config import settings 

337 

338 options = settings.SETTINGS_OVERRIDE 

339 return ( 

340 options.REFRESH_INTERVAL if interval is None else interval, 

341 options.REFRESHER_ENABLED if enabled is None else enabled, 

342 )

346async def settings_override_refresher( 

347 session_maker_factory: SessionMakerFactory, 

348 proxies: ProxyRegistry, 

349 interval: timedelta | None = None, 

350 *, 

351 enabled: bool | None = None, 

352 callbacks: CallbackRegistry | None = None, 

353) -> AsyncGenerator[None, None]: 

354 """Run the background override refresher for the duration of a lifespan. 

355 

356 Start the periodic refresher on enter (when ``enabled``), then cancel and 

357 drain it on exit. When ``enabled`` is ``False`` the context manager is a 

358 no-op and no refresh task is created -- the wrapped block still runs. 

359 

360 :param session_maker_factory: A zero-argument callable returning a 

361 service-scoped ``async_sessionmaker``, forwarded to 

362 :func:`start_refresh_task`. 

363 :param proxies: The wired proxy registry keyed by class identifier. 

364 :param interval: The wall-clock delay between refresh cycles. ``None``, the 

365 default, reads ``Settings.SETTINGS_OVERRIDE.REFRESH_INTERVAL``. 

366 :param enabled: Whether to start the background refresher. When ``False``, 

367 the context manager yields without creating a task. ``None``, the 

368 default, reads ``Settings.SETTINGS_OVERRIDE.REFRESHER_ENABLED``. 

369 :param callbacks: Optional rebind callbacks forwarded to 

370 :func:`start_refresh_task`, fired by the periodic loop when a watched 

371 override changes value. 

372 :return: None 

373 """ 

374 interval, enabled = resolve_refresher_options(interval, enabled=enabled) 

375 refresher: asyncio.Task | None = None 

376 if enabled: 

377 refresher = await start_refresh_task( 

378 session_maker_factory, proxies, interval, callbacks 

379 ) 

380 try: 

381 yield 

382 finally: 

383 if refresher is not None: 

384 refresher.cancel() 

385 with suppress(asyncio.CancelledError): 

386 await refresher
