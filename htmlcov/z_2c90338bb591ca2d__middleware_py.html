16"""Define middleware to manage temporary messages stored in cookies.""" 

17 

18__all__ = ["MessagesMiddleware"]

20import json 

21import logging 

22from collections import OrderedDict 

23from typing import ClassVar 

24 

25from itsdangerous import BadPayload, BadSignature 

26from pydantic import ValidationError 

27from starlette.middleware.base import BaseHTTPMiddleware, RequestResponseEndpoint 

28from starlette.requests import Request 

29from starlette.responses import Response 

30 

31from app.core.security import crypto_serializer 

32from app.core.utils import json_serializer 

33from app.sep.middleware.messages.models import Message

35logger = logging.getLogger("app.sep.middleware.messages") 

36

38class MessagesMiddleware(BaseHTTPMiddleware): 

39 """Manage temporary messages stored in cookies. 

40 

41 This middleware retrieves messages stored in a cookie from the incoming request, 

42 attaches them to the request state, and updates the response to include a cookie 

43 with the current message queue. If the cookie size exceeds a maximum threshold, 

44 older messages are discarded. 

45 

46 :cvar MAX_COOKIE_SIZE: Maximum allowed size (in bytes) for the messages cookie. 

47 Set to 1024. 

48 :vartype MAX_COOKIE_SIZE: int 

49 :cvar COOKIE_NAME: Name of the cookie used for storing messages. Set to "messages" 

50 :vartype COOKIE_NAME: str 

51 :cvar COOKIE_MAX_AGE: Maximum age (in seconds) for the messages cookie. Set to 

52 12 hours. 

53 :vartype COOKIE_MAX_AGE: int 

54 """ 

55 

56 MAX_COOKIE_SIZE: ClassVar[int] = 1024 

57 COOKIE_NAME: ClassVar[str] = "messages" 

58 COOKIE_MAX_AGE: ClassVar[int] = 60 * 60 * 12 

59 

60 async def dispatch( 

61 self, request: Request, call_next: RequestResponseEndpoint 

62 ) -> Response: 

63 """Retrieve messages from cookies and update the response. 

64 

65 This method parses the cookie (if present) to restore a list of messages into 

66 the request state. After processing the request, it serializes any remaining 

67 messages back into a cookie, ensuring that the cookie does not exceed the 

68 maximum allowed size. 

69 

70 :param request: The incoming HTTP request. 

71 :type request: Request 

72 :param call_next: The next callable in the middleware chain. 

73 :type call_next: RequestResponseEndpoint 

74 :return: The HTTP response with an updated messages cookie or with the cookie 

75 removed, in case no message is left in the queue. 

76 :rtype: Response 

77 """ 

78 request.state.messages = OrderedDict() 

79 if old_cookie := request.cookies.get(self.COOKIE_NAME): 

80 try: 

81 request.state.messages = OrderedDict.fromkeys( 

82 Message.model_validate(msg) 

83 for msg in json.loads(crypto_serializer.loads(old_cookie)) 

84 ) 

85 except (BadSignature, BadPayload, json.JSONDecodeError, ValidationError): 

86 logger.debug("Invalid messages cookie; treating as empty") 

87 

88 response = await call_next(request) 

89 while ( 

90 len( 

91 new_cookie := crypto_serializer.dumps( 

92 json_serializer( 

93 list(request.state.messages), 

94 encoders_kwargs={"exclude_none": True}, 

95 separators=(",", ":"), 

96 ) 

97 ) 

98 ) 

99 > self.MAX_COOKIE_SIZE 

100 ) and request.state.messages: 

101 discarded_msg, _ = request.state.messages.popitem(last=False) 

102 logger.debug("Discarding message %s for %s", discarded_msg, request.client) 

103 

104 if request.state.messages: 

105 response.set_cookie( 

106 key=self.COOKIE_NAME, 

107 value=new_cookie, 

108 httponly=True, 

109 max_age=self.COOKIE_MAX_AGE, 

110 ) 

111 else: 

112 response.delete_cookie(self.COOKIE_NAME) 

113 

114 return response
