16"""Define the constrained syncer fields and the values their config accepts. 

17 

18A syncer threshold is configurable three ways — a ``SYNCERS[]`` entry in 

19``settings.yaml``, JSON in ``SEP__SYNCERS``, or a ``SEP__SYNCER_EXTRA_KWARGS__<KEY>`` 

20env leaf — and ``SyncOptions`` forwards every one of them verbatim, untyped. The 

21syncer field and the load-time configuration check therefore have to agree on the 

22accepted values, so the annotated type, the field name and the spelling quoted back to 

23an operator all live here, one edit apart. This module deliberately imports nothing 

24from the SEP app so ``app.sep.config`` can use it. 

25""" 

26 

27from datetime import timedelta 

28from typing import Annotated, Any, Final, NamedTuple 

29 

30from annotated_types import Ge, Gt 

31from pydantic import TypeAdapter 

32 

33StaleRunAfter = Annotated[timedelta, Gt(timedelta(0))] 

34"""Define the age beyond which an idle in-progress sync run is reclaimable. 

35 

36Positivity is an annotation constraint rather than a ``field_validator`` because 

37``SyncOptions`` carries ``extra="allow"`` and forwards every extra key verbatim, and 

38runtime-override coercion re-checks annotated-type constraints but does not re-run 

39field validators. 

40""" 

41 

42MissingGraceGenerations = Annotated[int, Ge(2)] 

43"""Define how many complete generations must report an entity absent before removal. 

44 

45The floor is 2, not 1: at 1 the grace counter collapses back to acting on a single 

46reported absence, which is the behaviour it exists to end. Expressed as an annotation 

47constraint for the reason :data:`StaleRunAfter` is. 

48""" 

49 

50 

51class SyncerFieldConstraint(NamedTuple): 

52 """Pair a constrained syncer field with the spelling of its accepted values. 

53 

54 :param adapter: Validator for the field's annotated type, constraints included. 

55 :param accepted: Human-readable list of the forms the field accepts, quoted back 

56 to the operator when their value is refused. 

57 """ 

58 

59 adapter: TypeAdapter[Any] 

60 accepted: str 

61 

62 

63CONSTRAINED_SYNCER_FIELDS: Final[dict[str, SyncerFieldConstraint]] = { 

64 "stale_run_after": SyncerFieldConstraint( 

65 TypeAdapter(StaleRunAfter), 

66 "integer or float seconds (3600), an ISO-8601 duration (PT1H), or HH:MM:SS", 

67 ), 

68 "missing_grace_generations": SyncerFieldConstraint( 

69 TypeAdapter(MissingGraceGenerations), 

70 "an integer of 2 or more", 

71 ), 

72} 

73"""Map each constrained syncer field to the validator its configured value must pass. 

74 

75``SyncOptions`` and ``SyncerExtraKwargs`` both allow untyped extras, so a value 

76configured for one of these fields is only checked against the field's real type once 

77a syncer is constructed — which, for the request-scoped ``get_syncers`` dependency, 

78means once per request. Checking it against this mapping at settings load turns a 

79misconfiguration into a single startup failure instead. Keys are the field names as 

80:class:`~app.core.models.BaseLowercaseModel` stores them, lowercased. 

81"""
